cmake_minimum_required(VERSION 4.4)
project(teaql_toolchain_probe LANGUAGES CXX)
add_library(teaql_probe)
target_compile_features(teaql_probe PUBLIC cxx_std_20)
set(TEAQL_SOURCE_DIR "" CACHE PATH "TEAQL repository checkout")
if(NOT EXISTS "${TEAQL_SOURCE_DIR}/include/teaql/memory/reference.hpp")
    message(FATAL_ERROR "Set TEAQL_SOURCE_DIR to the repository checkout")
endif()
target_include_directories(teaql_probe PRIVATE "${TEAQL_SOURCE_DIR}/include")
target_sources(teaql_probe PUBLIC FILE_SET CXX_MODULES FILES probe.cppm)
add_executable(teaql_probe_consumer consumer.cpp)
target_link_libraries(teaql_probe_consumer PRIVATE teaql_probe)
set_target_properties(teaql_probe_consumer PROPERTIES CXX_SCAN_FOR_MODULES ON)
include(CTest)
add_test(NAME imported_existing_types COMMAND teaql_probe_consumer)
