add_executable(teaql_build_consumer main.cpp)
target_compile_features(teaql_build_consumer PRIVATE cxx_std_20)
set_target_properties(teaql_build_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(TEAQL_ENABLE_MODULES)
    target_compile_definitions(teaql_build_consumer PRIVATE TEAQL_BUILD_USE_MODULES=1)

    target_link_libraries(teaql_build_consumer PRIVATE teaql::provider_sqlite_module)
else()
    set_target_properties(teaql_build_consumer PROPERTIES CXX_SCAN_FOR_MODULES OFF)
    target_link_libraries(teaql_build_consumer PRIVATE teaql::provider_sqlite)
endif()
