cmake_minimum_required(VERSION 4.4)
project(teaql_external_consumer LANGUAGES CXX)

option(TEAQL_CONSUME_MODULES "Use the installed named-module frontend" ON)
if(TEAQL_CONSUME_MODULES)
    find_package(teaql_cpp CONFIG REQUIRED COMPONENTS modules)
else()
    find_package(teaql_cpp CONFIG REQUIRED)
endif()
add_executable(teaql_external_consumer main.cpp header_bridge.cpp)
target_compile_features(teaql_external_consumer PRIVATE cxx_std_20)
set_target_properties(teaql_external_consumer PROPERTIES CXX_EXTENSIONS OFF)
if(TEAQL_CONSUME_MODULES)
    target_compile_definitions(teaql_external_consumer PRIVATE TEAQL_CONSUME_MODULES=1)
    set_target_properties(teaql_external_consumer PROPERTIES CXX_SCAN_FOR_MODULES ON)
    target_link_libraries(teaql_external_consumer PRIVATE teaql::provider_sqlite_module)
else()
    set_target_properties(teaql_external_consumer PROPERTIES CXX_SCAN_FOR_MODULES OFF)
    target_link_libraries(teaql_external_consumer PRIVATE teaql::provider_sqlite)
endif()
enable_testing()
add_test(NAME installed_consumer_contract COMMAND teaql_external_consumer)
